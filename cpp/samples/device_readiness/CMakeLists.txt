cmake_minimum_required(VERSION 3.22.1)
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SmartSpectraDeviceReadiness CXX)
endif ()
if (NOT TARGET SmartSpectra::SDK)
    find_package(SmartSpectra CONFIG REQUIRED)
endif ()
add_executable(device_readiness main.cpp)
target_compile_features(device_readiness PRIVATE cxx_std_20)
target_link_libraries(device_readiness PRIVATE SmartSpectra::SDK)

add_executable(device_readiness_advanced advanced.cpp)
target_compile_features(device_readiness_advanced PRIVATE cxx_std_20)
target_link_libraries(device_readiness_advanced PRIVATE SmartSpectra::SDK)

include(CTest)
if (BUILD_TESTING)
    add_executable(readiness_presentation_test presentation_test.cpp)
    target_compile_features(readiness_presentation_test PRIVATE cxx_std_20)
    target_link_libraries(readiness_presentation_test PRIVATE SmartSpectra::SDK)
    add_test(NAME readiness_presentation COMMAND readiness_presentation_test)
endif ()
